	\section{Predicate Definitions}
	\label{sec:predicate-definitions}
	\idx{Predicate Definitions}
	
	\setcounter{defcounter}{0}
	\begin{predicatetable}
		
		% ============================================
		% TRANSFORMER ARCHITECTURE
		% ============================================
		\phantomsection\defnum\label{def:transformer} $\IsTransformer{T}$ & 
		$\IsTransformer{T} \equiv T = (E, D, H, L,\; \mathsf{Block})$ \\ \hdashline
		
		\phantomsection\defnum\label{def:embedding} $\IsEmbedding{E}{d}$ & 
		$\IsEmbedding{E}{d} \equiv E: \mathsf{Vocab} \to \R^d$ \\ \hdashline
		
		\phantomsection\defnum\label{def:positional-encoding} $\mathsf{PE}(pos, i)$ & 
		$\begin{array}{r@{}l}
		\mathsf{PE}(pos, 2i) &\equiv \sin\left(\frac{pos}{10000^{2i/d}}\right) \\
		\mathsf{PE}(pos, 2i+1) &\equiv \cos\left(\frac{pos}{10000^{2i/d}}\right)
		\end{array}$ \\ \hdashline
		
		\phantomsection\defnum\label{def:self-attention} $\mathsf{SelfAttention}(Q,K,V)$ & 
		$\mathsf{SelfAttention}(Q,K,V) \equiv \mathsf{softmax}\left(\frac{QK^{\mathsf{T}}}{\sqrt{d_k}}\right) V$ \\ \hdashline
		
		\phantomsection\defnum\label{def:attention-head} $\mathsf{AttHead}_h(Q,K,V)$ & 
		$\mathsf{AttHead}_h(Q,K,V) \equiv \mathsf{SelfAttention}(QW_h^Q, KW_h^K, VW_h^V)$ \\ \hdashline
		
		\phantomsection\defnum\label{def:multi-head-attention} $\mathsf{MHA}(Q,K,V)$ & 
		$\mathsf{MHA}(Q,K,V) \equiv \mathsf{Concat}(\mathsf{AttHead}_1,\ldots,\mathsf{AttHead}_H) W^O$ \\ \hdashline
		
		\phantomsection\defnum\label{def:feed-forward} $\mathsf{FFN}(x)$ & 
		$\mathsf{FFN}(x) \equiv \max(0, xW_1 + b_1) W_2 + b_2$ \\ \hdashline
		
		\phantomsection\defnum\label{def:gelu} $\mathsf{GELU}(x)$ & 
		$\mathsf{GELU}(x) \equiv x \cdot \Phi(x) = \frac{x}{2}\left(1 + \operatorname{erf}\left(\frac{x}{\sqrt{2}}\right)\right)$ \\ \hdashline
		
		\phantomsection\defnum\label{def:layer-norm} $\mathsf{LayerNorm}(x)$ & 
		$\mathsf{LayerNorm}(x) \equiv \frac{x - \mu}{\sigma} \odot \gamma + \beta$ \\ \hdashline
		
		\phantomsection\defnum\label{def:transformer-block} $\mathsf{Block}(x)$ & 
		$\begin{array}{r@{}l}
		\mathsf{Block}(x) \equiv & x' = \mathsf{LayerNorm}(x + \mathsf{MHA}(x,x,x)) \\
		& \mathsf{LayerNorm}(x' + \mathsf{FFN}(x'))
		\end{array}$ \\ \hdashline
		
		\phantomsection\defnum\label{def:decoder-only} $\mathsf{DecoderOnly}(T)$ & 
		$\mathsf{DecoderOnly}(T) \equiv \Forall i,\; \mathsf{Block}_i \text{ uses masked self-attention}$ \\ \hdashline
		
		\phantomsection\defnum\label{def:softmax} $\mathsf{softmax}(z)$ & 
		$\mathsf{softmax}(z) \equiv \frac{e^{z_i}}{\sum_{j=1}^C e^{z_j}}$ \\ \hdashline
		
		% ============================================
		% TRAINING CONCEPTS
		% ============================================
		\phantomsection\defnum\label{def:cross-entropy} $\mathsf{CrossEntropy}(y, \hat{y})$ & 
		$\mathsf{CrossEntropy}(y, \hat{y}) \equiv -\sum_{i=1}^C y_i \log(\hat{y}_i)$ \\ \hdashline
		
		\phantomsection\defnum\label{def:entropy} $\mathsf{Entropy}(y)$ & 
		$\mathsf{Entropy}(y) \equiv -\sum_{i=1}^C y_i \log(y_i)$ \\ \hdashline
		
		\phantomsection\defnum\label{def:kl-divergence} $\mathsf{KL}(y \| \hat{y})$ & 
		$\mathsf{KL}(y \| \hat{y}) \equiv \sum_{i=1}^C y_i \log\left(\frac{y_i}{\hat{y}_i}\right)$ \\ \hdashline
		
		\phantomsection\defnum\label{def:cross-entropy-relation} $\mathsf{CrossEntropyRel}$ & 
		$\mathsf{CrossEntropy}(y, \hat{y}) = \mathsf{Entropy}(y) + \mathsf{KL}(y \| \hat{y})$ \\ \hdashline
		
		\phantomsection\defnum\label{def:gradient} $\mathsf{Gradient}_{\theta}(\mathcal{L})$ & 
		$\mathsf{Gradient}_{\theta}(\mathcal{L}) \equiv \frac{\partial \mathcal{L}}{\partial \theta}$ \\ \hdashline
		
		\phantomsection\defnum\label{def:backpropagation} $\mathsf{Backprop}(\mathcal{L}, \theta)$ & 
		$\mathsf{Backprop}(\mathcal{L}, \theta) \equiv \theta \leftarrow \theta - \eta \nabla_{\theta} \mathcal{L}$ \\ \hdashline
		
		\phantomsection\defnum\label{def:adam-optimizer} $\mathsf{Adam}(\theta, \nabla \mathcal{L})$ & 
		$\begin{array}{r@{}l}
		\mathsf{Adam}(\theta, \nabla \mathcal{L}) \equiv & m_t = \beta_1 m_{t-1} + (1-\beta_1) \nabla \mathcal{L} \\
		& v_t = \beta_2 v_{t-1} + (1-\beta_2) (\nabla \mathcal{L})^2 \\
		& \hat{m}_t = \frac{m_t}{1-\beta_1^t},\; \hat{v}_t = \frac{v_t}{1-\beta_2^t} \\
		& \theta \leftarrow \theta - \eta \frac{\hat{m}_t}{\sqrt{\hat{v}_t} + \epsilon}
		\end{array}$ \\ \hdashline
		
		\phantomsection\defnum\label{def:batch} $\mathsf{Batch}_b(D)$ & 
		$\mathsf{Batch}_b(D) \equiv \{ (x_i, y_i) \}_{i=1}^b \subseteq D$ \\ \hdashline
		
		\phantomsection\defnum\label{def:epoch} $\mathsf{Epoch}(D, b)$ & 
		$\mathsf{Epoch}(D, b) \equiv \bigcup_{j=1}^{\lceil |D|/b \rceil} \mathsf{Batch}_b(D_j)$ \\ \hdashline
		
		\phantomsection\defnum\label{def:learning-rate-schedule} $\eta(t)$ & 
		$\eta(t) \equiv \eta_0 \cdot \min\left(1, \frac{t}{t_{\text{warmup}}}\right) \cdot \frac{1}{\sqrt{\max(t, t_{\text{warmup}})}}$ \\ \hdashline
		
		\phantomsection\defnum\label{def:loss-function} $\mathcal{L}(\theta)$ & 
		$\mathcal{L}(\theta) \equiv \frac{1}{N} \sum_{i=1}^N \mathsf{CrossEntropy}(y_i, f_{\theta}(x_i))$ \\ \hdashline
		
		\phantomsection\defnum\label{def:convergence} $\mathsf{Converges}(\mathcal{L}_t)$ & 
		$\mathsf{Converges}(\mathcal{L}_t) \equiv \Exists \mathcal{L}^*,\; \lim_{t \to \infty} \mathcal{L}_t = \mathcal{L}^*$ \\ \hdashline
		
		% ============================================
		% OPTIMIZATION PREDICATES
		% ============================================
		\phantomsection\defnum\label{def:minimizer} $\mathsf{Minimizer}(\hat{y}, \mathcal{L})$ & 
		$\mathsf{Minimizer}(\hat{y}, \mathcal{L}) \equiv \Forall \hat{y}',\; \mathcal{L}(\hat{y}') \ge \mathcal{L}(\hat{y})$ \\ \hdashline
		
		\phantomsection\defnum\label{def:exp-loss} $\mathsf{ExpLoss}(\hat{y}, x)$ & 
		$\mathsf{ExpLoss}(\hat{y}, x) \equiv \sum_{y \in \{0,1\}} P(y|x) \cdot \ell_{01}(y, \hat{y})$ \\ \hdashline
		
		\phantomsection\defnum\label{def:bayes-optimal} $\mathsf{BayesOptimal}(\hat{y}, x)$ & 
		$\mathsf{BayesOptimal}(\hat{y}, x) \equiv \mathsf{Minimizer}(\hat{y}, \mathsf{ExpLoss}(\cdot, x))$ \\ \hdashline
		
		\phantomsection\defnum\label{def:zero-one-loss} $\ell_{01}(y, \hat{y})$ & 
		$\ell_{01}(y, \hat{y}) \equiv \begin{cases} 0 & y = \hat{y} \\ 1 & y \neq \hat{y} \end{cases}$ \\ \hdashline
		
		\phantomsection\defnum\label{def:ce-minimizer} $\mathsf{CEMinimizer}(\hat{y}, y)$ & 
		$\mathsf{CEMinimizer}(\hat{y}, y) \equiv \Forall \hat{y}',\; \mathsf{CrossEntropy}(y, \hat{y}') \ge \mathsf{CrossEntropy}(y, \hat{y})$ \\ \hdashline

		% ============================================
		\phantomsection\defnum\label{def:tp} $\mathsf{TP}$ &

		\phantomsection\defnum\label{def:tn} $\mathsf{TN}$ &

		\phantomsection\defnum\label{def:fn} $\mathsf{FN}$ &

		\phantomsection\defnum\label{def:accuracy} $\mathsf{Accuracy}$ &

		\phantomsection\defnum\label{def:f1-score} $\mathsf{F1}$ &

		\phantomsection\defnum\label{def:tpr} $\mathsf{TPR}(\tau)$ & 
		$\mathsf{TPR}(\tau) \equiv \frac{\mathsf{TP}(\tau)}{\mathsf{TP}(\tau) + \mathsf{FN}(\tau)}$ \\ \hdashline
		
		\phantomsection\defnum\label{def:fpr} $\mathsf{FPR}(\tau)$ & 
		$\mathsf{FPR}(\tau) \equiv \frac{\mathsf{FP}(\tau)}{\mathsf{FP}(\tau) + \mathsf{TN}(\tau)}$ \\ \hdashline

		\phantomsection\defnum\label{def:auc} $\mathsf{AUC}$ &

	\subsection{Planetary System Encoding}
	\label{subsec:encoding-algorithm}
	\idx{encoding algorithm}
	
	The following procedures transform planetary systems into sequences of tokens suitable for transformer processing, as described in Definitions \ref{def:encoding-grid}, \ref{def:character-map}, and \ref{def:word}.
	
	\begin{algorithmic}[1]
		
		\Procedure{EncodeSystem}{$S, N$} \Comment{Def. \ref{def:encoding-grid}, \ref{def:character-map}}
		\State $\log a_{\min} \gets -3,\; \log a_{\max} \gets 3$ \Comment{Def. \ref{def:encoding-bounds}}
		\State $\log m_{\min} \gets -4,\; \log m_{\max} \gets 4$
		\State $\Delta a \gets (\log a_{\max} - \log a_{\min}) / N$
		\State $\Delta m \gets (\log m_{\max} - \log m_{\min}) / N$
		\State $\mathsf{word} \gets []$
		\For{each $p \in S$}
		\State $\ell_a \gets \log_{10}(\mathsf{A}_{a}(p))$ \Comment{Def. \ref{def:semi-major-axis}}
		\State $\ell_m \gets \log_{10}(\mathsf{Mass}_{M}(p))$ \Comment{Def. \ref{def:mass}}
		\State $i \gets \min\left(\left\lfloor \frac{\ell_a - \log a_{\min}}{\Delta a} \right\rfloor + 1,\; N\right)$
		\State $j \gets \min\left(\left\lfloor \frac{\ell_m - \log m_{\min}}{\Delta m} \right\rfloor + 1,\; N\right)$
		\State $c_{ij} \gets \mathsf{CharMap}_{N}(p)$ \Comment{Def. \ref{def:character-map}}
		\State $\mathsf{word}.\mathsf{append}(c_{ij})$
		\EndFor
		\State \Return $\mathsf{word}$ \Comment{Def. \ref{def:word}}
		\EndProcedure
		
		\Statex
		
		\Procedure{DecodeSystem}{$\mathsf{word}, N$} \Comment{Inverse of Def. \ref{def:character-map}}
		\State $\log a_{\min} \gets -3,\; \log a_{\max} \gets 3$
		\State $\log m_{\min} \gets -4,\; \log m_{\max} \gets 4$
		\State $\Delta a \gets (\log a_{\max} - \log a_{\min}) / N$
		\State $\Delta m \gets (\log m_{\max} - \log m_{\min}) / N$
		\State $S \gets []$
		\For{each $c_{ij} \in \mathsf{word}$}
		\State $\ell_a \gets \log a_{\min} + (i-1)\Delta a + \frac{\Delta a}{2}$
		\State $\ell_m \gets \log m_{\min} + (j-1)\Delta m + \frac{\Delta m}{2}$
		\State $a_p \gets 10^{\ell_a},\; m_p \gets 10^{\ell_m}$
		\State $p \gets \mathsf{Planet}(m_p, a_p)$ \Comment{Def. \ref{def:planet}}
		\State $S.\mathsf{append}(p)$
		\EndFor
		\State \Return $\mathsf{Sys}_{|S|}(S)$ \Comment{Def. \ref{def:system}}
		\EndProcedure
		
	\end{algorithmic}
	
	\subsection{Transformer Training Procedure}
	\label{subsec:training-algorithm}
	\idx{training algorithm}
	
	The training procedure for the transformer model on planetary system data, implementing the loss function (Def. \ref{def:loss-function}) and optimization (Def. \ref{def:adam-optimizer}).
	
	\begin{algorithmic}[1]
		
		\Procedure{ComputeLoss}{$\theta, X, Y$} \Comment{Def. \ref{def:loss-function}, \ref{def:cross-entropy}}
		\State $X_{\text{enc}} \gets \mathsf{Embedding}(X) + \mathsf{PE}(X)$ \Comment{Def. \ref{def:embedding}, \ref{def:positional-encoding}}
		\For{$l = 1$ to $L$}
		\State $X_{\text{enc}} \gets \mathsf{LayerNorm}(X_{\text{enc}} + \mathsf{MHA}(X_{\text{enc}}, X_{\text{enc}}, X_{\text{enc}}))$ \Comment{Def. \ref{def:layer-norm}, \ref{def:multi-head-attention}}
		\State $X_{\text{enc}} \gets \mathsf{LayerNorm}(X_{\text{enc}} + \mathsf{FFN}(X_{\text{enc}}))$ \Comment{Def. \ref{def:feed-forward}}
		\EndFor
		\State $\hat{Y} \gets \mathsf{softmax}(X_{\text{enc}} W_{\text{out}})$ \Comment{Def. \ref{def:softmax}}
		\State $\mathcal{L} \gets \frac{1}{|X|} \sum_{i=1}^{|X|} \mathsf{CrossEntropy}(Y_i, \hat{Y}_i)$
		\State \Return $\mathcal{L}$
		\EndProcedure
		
		\Statex
		
		\Procedure{TrainTransformer}{$D, \theta, \eta(t), b, E$} \Comment{Def. \ref{def:adam-optimizer}, \ref{def:convergence}}
		\For{$epoch = 1$ to $E$} \Comment{Def. \ref{def:epoch}}
		\State $\mathsf{shuffle}(D)$
		\For{each batch $(X, Y) \in \mathsf{Batch}_b(D)$} \Comment{Def. \ref{def:batch}}
		\State $\mathcal{L}_{\text{batch}} \gets \mathsf{ComputeLoss}(\theta, X, Y)$
		\State $\nabla \mathcal{L} \gets \mathsf{Gradient}_{\theta}(\mathcal{L}_{\text{batch}})$ \Comment{Def. \ref{def:gradient}}
		\State $\theta \gets \mathsf{Adam}(\theta, \nabla \mathcal{L})$ \Comment{Def. \ref{def:adam-optimizer}}
		\EndFor
		\If{$\mathsf{Converges}(\mathcal{L}_t)$} \Comment{Def. \ref{def:convergence}}
		\State \textbf{break}
		\EndIf
		\EndFor
		\State \Return $\theta$
		\EndProcedure
		
	\end{algorithmic}
	
	\subsection{Planetary System Generation}
	\label{subsec:generation-algorithm}
	\idx{generation algorithm}
	
	The autoregressive generation of planetary systems using the trained transformer, where each new planet is predicted conditioned on previously generated planets (Theorem \ref{thm:cross-entropy-minimization}).
	
	\begin{algorithmic}[1]
		
		\Procedure{GenerateNextToken}{$T, \theta, \mathsf{context}, \tau$} \Comment{Autoregressive prediction}
		\State $X_{\text{enc}} \gets \mathsf{Embedding}(\mathsf{context}) + \mathsf{PE}(\mathsf{context})$
		\For{$l = 1$ to $L$}
		\State $X_{\text{enc}} \gets \mathsf{LayerNorm}(X_{\text{enc}} + \mathsf{MaskedMHA}(X_{\text{enc}}, X_{\text{enc}}, X_{\text{enc}}))$
		\State $X_{\text{enc}} \gets \mathsf{LayerNorm}(X_{\text{enc}} + \mathsf{FFN}(X_{\text{enc}}))$
		\EndFor
		\State $\mathbf{z} \gets X_{\text{enc}}[-1] W_{\text{out}}$ \Comment{Last position logits}
		\State $p_i \gets \dfrac{\exp(z_i / \tau)}{\sum_j \exp(z_j / \tau)}$ \Comment{Temperature sampling}
		\State $c \sim \mathsf{Categorical}(p)$
		\State \Return $c$
		\EndProcedure
		
		\Statex
		
		\Procedure{GenerateSystem}{$T, \theta, n_{\max}, \tau$} \Comment{Def. \ref{def:word}, \ref{def:system}}
		\State $\mathsf{word} \gets []$
		\State $t \gets 1$
		\While{$t \le n_{\max}$}
		\State $c \gets \mathsf{GenerateNextToken}(T, \theta, \mathsf{word}, \tau)$
		\If{$c = \mathsf{EOS}$} \Comment{End-of-sequence token}
		\State \textbf{break}
		\EndIf
		\State $\mathsf{word}.\mathsf{append}(c)$
		\State $t \gets t + 1$
		\EndWhile
		\State $S \gets \mathsf{DecodeSystem}(\mathsf{word}, N)$
		\State \Return $S$
		\EndProcedure
		
		\Statex
		
		\Procedure{ComputeStability}{$S, M_*$} \Comment{Def. \ref{def:stability}, \ref{def:chaotic}}
		\State $\mathsf{stable} \gets \mathsf{True}$
		\For{$i = 1$ to $|S|-1$}
		\State $R_{\mathrm{Hill}} \gets \left(\dfrac{m_i + m_{i+1}}{3M_*}\right)^{1/3} \left(\dfrac{a_i + a_{i+1}}{2}\right)$ \Comment{Def. \ref{def:hill-radius}}
		\State $\Delta \gets \dfrac{a_{i+1} - a_i}{R_{\mathrm{Hill}}}$
		\If{$\Delta \le 2\sqrt{3}$} \Comment{Def. \ref{def:critical-stability}}
		\State $\mathsf{stable} \gets \mathsf{False}$ \Comment{Theorem \ref{thm:hill-stability}}
		\EndIf
		\EndFor
		\State \Return $\mathsf{stable}$
		\EndProcedure
		
	\end{algorithmic}

	\subsection*{Proof of \ref{thm:cross-entropy-minimization}. (CrossEntropyMin, \cite{vaswani2017attention})}
	\idx{proof of cross-entropy minimization}
	\begin{prooftable}
		1 & $\begin{array}{r@{}l}\vdash & \mathsf{CrossEntropy}(y, \hat{y}) = -\sum_{i=1}^C y_i \log(\hat{y}_i)\end{array}$ & Def. \ref{def:cross-entropy} \\
		2 & $\begin{array}{r@{}l}\vdash & \Exists k,\; y_k = 1 \;\land\; \Forall i \neq k,\; y_i = 0\end{array}$ & $\top$ \\
		3 & $\begin{array}{r@{}l}\vdash & \mathsf{CrossEntropy}(y, \hat{y}) = -\log(\hat{y}_k)\end{array}$ & $\top$ \\
		4 & $\begin{array}{r@{}l}\vdash & \sum_{i=1}^C \hat{y}_i = 1 \;\land\; \Forall i,\; \hat{y}_i \ge 0\end{array}$ & Def. \ref{def:softmax} \\
		5 & $\begin{array}{r@{}l}\vdash & \mathsf{KL}(y \| \hat{y}) = \sum_{i=1}^C y_i \log\left(\frac{y_i}{\hat{y}_i}\right) \ge 0\end{array}$ & $\top$ \\
		6 & $\begin{array}{r@{}l}\vdash & \mathsf{CrossEntropy}(y, \hat{y}) = \mathsf{KL}(y \| \hat{y}) - \sum_{i=1}^C y_i \log(y_i)\end{array}$ & $\top$ \\
		7 & $\begin{array}{r@{}l}\vdash & -\sum_{i=1}^C y_i \log(y_i) = 0\end{array}$ & $\top$ \\
		8 & $\begin{array}{r@{}l}\vdash & \mathsf{CrossEntropy}(y, \hat{y}) \ge 0\end{array}$ & $\top$ \\
		9 & $\begin{array}{r@{}l}\vdash & \mathsf{CrossEntropy}(y, \hat{y}) = 0 \;\leftrightarrow\; \mathsf{KL}(y \| \hat{y}) = 0\end{array}$ & $\top$ \\
		10 & $\begin{array}{r@{}l}\vdash & \mathsf{KL}(y \| \hat{y}) = 0 \;\leftrightarrow\; \Forall i,\; y_i = \hat{y}_i\end{array}$ & $\top$ \\
		11 & $\begin{array}{r@{}l}\vdash & \hat{y} = y \;\rightarrow\; \mathsf{CEMinimizer}(\hat{y}, y)\end{array}$ & Def. \ref{def:ce-minimizer} \\
	\end{prooftable}
	
	\subsection*{Proof of \ref{thm:auc-properties}. (AUCProperties, \cite{vaswani2017attention})}
	\idx{proof of AUC properties}
	\begin{prooftable}
		1 & $\begin{array}{r@{}l}\vdash & \mathsf{AUC} = \int_0^1 \mathsf{TPR}(\tau) \, d\mathsf{FPR}(\tau)\end{array}$ & Def. \ref{def:auc} \\
		2 & $\begin{array}{r@{}l}\vdash & 0 \le \mathsf{TPR}(\tau) \le 1 \;\land\; 0 \le \mathsf{FPR}(\tau) \le 1\end{array}$ & $\top$ \\
		3 & $\begin{array}{r@{}l}\vdash & 0 \le \mathsf{AUC} \le \int_0^1 1 \, d\mathsf{FPR} = 1\end{array}$ & $\top$ \\
		4 & $\begin{array}{r@{}l}\vdash & \mathsf{TPR}_{\text{rand}}(\tau) = \mathsf{FPR}_{\text{rand}}(\tau)\end{array}$ & $\top$ \\
		5 & $\begin{array}{r@{}l}\vdash & \mathsf{AUC}_{\text{rand}} = \int_0^1 t \, dt = \frac{1}{2}\end{array}$ & $\top$ \\
		6 & $\begin{array}{r@{}l}\vdash & \mathsf{TPR}_{\text{perfect}}(\tau) = 1 \text{ for } \tau \le 1,\; \mathsf{FPR}_{\text{perfect}}(\tau) = 0 \text{ for } \tau > 0\end{array}$ & $\top$ \\
		7 & $\begin{array}{r@{}l}\vdash & \mathsf{AUC}_{\text{perfect}} = \int_0^1 1 \, d\mathsf{FPR} = 1\end{array}$ & $\top$ \\
	\end{prooftable}
	
	\subsection*{Proof of \ref{thm:bayes-optimal}. (BayesOptimal, \cite{vaswani2017attention})}
	\idx{proof of Bayes optimal classifier}
	\begin{prooftable}
		1 & $\begin{array}{r@{}l}\vdash & \eta(x) = P(y=1|x)\end{array}$ & $\top$ \\
		2 & $\begin{array}{r@{}l}\vdash & \mathsf{ExpLoss}(\hat{y}, x) = \eta(x) \ell_{01}(1, \hat{y}) + (1-\eta(x)) \ell_{01}(0, \hat{y})\end{array}$ & Def. \ref{def:exp-loss} \\
		3 & $\begin{array}{r@{}l}\vdash & \hat{y}=1 \;\rightarrow\; \mathsf{ExpLoss}(1, x) = 1 - \eta(x)\end{array}$ & $\top$ \\
		4 & $\begin{array}{r@{}l}\vdash & \hat{y}=0 \;\rightarrow\; \mathsf{ExpLoss}(0, x) = \eta(x)\end{array}$ & $\top$ \\
		5 & $\begin{array}{r@{}l}\vdash & 1 - \eta(x) < \eta(x) \;\leftrightarrow\; \eta(x) > \frac{1}{2}\end{array}$ & $\top$ \\
		6 & $\begin{array}{r@{}l}\vdash & \eta(x) > \frac{1}{2} \;\rightarrow\; \mathsf{Minimizer}(1, \mathsf{ExpLoss}(\cdot, x))\end{array}$ & Def. \ref{def:minimizer} \\
		7 & $\begin{array}{r@{}l}\vdash & \eta(x) > \frac{1}{2} \;\rightarrow\; \mathsf{BayesOptimal}(1, x)\end{array}$ & Def. \ref{def:bayes-optimal} \\
	\end{prooftable}
	
	\subsection*{Proof of \ref{thm:hill-stability}. (HillStability, \cite{ngpps2020})}
	\idx{proof of Hill stability criterion}
	\begin{prooftable}
		1 & $\begin{array}{r@{}l}\vdash & \mathsf{R}_{\mathrm{Hill}}(p_1, p_2, M_*) = \left(\frac{m_1+m_2}{3M_*}\right)^{1/3} \left(\frac{a_1+a_2}{2}\right)\end{array}$ & Def. \ref{def:hill-radius} \\
		2 & $\begin{array}{r@{}l}\vdash & \Delta = \frac{a_2 - a_1}{\mathsf{R}_{\mathrm{Hill}}(p_1, p_2, M_*)}\end{array}$ & $\top$ \\
		3 & $\begin{array}{r@{}l}\vdash & \Delta_c = 2\sqrt{3}\end{array}$ & Def. \ref{def:critical-stability} \\
		4 & $\begin{array}{r@{}l}\vdash & \Delta > \Delta_c \;\rightarrow\; \neg \mathsf{OrbitCross}(p_1, p_2)\end{array}$ & $\top$ \\
		5 & $\begin{array}{r@{}l}\vdash & \Delta \le \Delta_c \;\rightarrow\; \mathsf{OrbitCross}(p_1, p_2)\end{array}$ & $\top$ \\
		6 & $\begin{array}{r@{}l}\vdash & \Delta > 2\sqrt{3} \;\leftrightarrow\; \neg \mathsf{OrbitCross}(p_1, p_2)\end{array}$ & $\top$ \\
	\end{prooftable}
	
	\subsection*{Proof of \ref{thm:encoding-bijection}. (EncodingBijection, \cite{ngpps2020})}
	\idx{proof of encoding bijection}
	\begin{prooftable}
		1 & $\begin{array}{r@{}l}\vdash & \mathsf{Grid}_{N}^{r}(G) = \bigcup_{i=1}^N \bigcup_{j=1}^N [x_i, x_{i+1}] \times [y_j, y_{j+1}]\end{array}$ & Def. \ref{def:encoding-grid} \\
		2 & $\begin{array}{r@{}l}\vdash & x_i = \log a_{\min} + (i-1) \frac{\log a_{\max} - \log a_{\min}}{N}\end{array}$ & Def. \ref{def:encoding-bounds} \\
		3 & $\begin{array}{r@{}l}\vdash & y_j = \log m_{\min} + (j-1) \frac{\log m_{\max} - \log m_{\min}}{N}\end{array}$ & Def. \ref{def:encoding-bounds} \\
		4 & $\begin{array}{r@{}l}\vdash & \Forall i \neq i',\; [x_i, x_{i+1}] \cap [x_{i'}, x_{i'+1}] = \emptyset\end{array}$ & $\top$ \\
		5 & $\begin{array}{r@{}l}\vdash & \Forall j \neq j',\; [y_j, y_{j+1}] \cap [y_{j'}, y_{j'+1}] = \emptyset\end{array}$ & $\top$ \\
		6 & $\begin{array}{r@{}l}\vdash & \Forall p,\; \Exists! (i,j),\; (\log a_p, \log m_p) \in [x_i, x_{i+1}] \times [y_j, y_{j+1}]\end{array}$ & $\top$ \\
		7 & $\begin{array}{r@{}l}\vdash & \mathsf{CharMap}_{N}(p) = c_{ij} \text{ where } (\log a_p, \log m_p) \in \mathsf{Grid}_{N}^{r}\end{array}$ & Def. \ref{def:character-map} \\
		8 & $\begin{array}{r@{}l}\vdash & \mathsf{CharMap}_{N}(p) = \mathsf{CharMap}_{N}(q) \;\leftrightarrow\; p,q \in \mathsf{GridCell}_{ij}\end{array}$ & $\top$ \\
	\end{prooftable}
